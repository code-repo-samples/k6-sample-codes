/**
 * =============================================================================
 * High-Throughput k6 Performance Testing Framework
 * =============================================================================
 *
 * REPORT STRUCTURE
 * ----------------
 *
 * 1. Overall Summary
 *
 * 2. Phase-wise Breakdown
 *      Phase
 *        - Target TPS
 *        - Actual TPS
 *        - Transactions (logical requests, excl. redirect hops)
 *        - Requests (raw HTTP requests, incl. redirect hops)
 *        - Dropped iterations (informational only -- does NOT fail the run)
 *        - Error %  (fails the phase/run only above the configured threshold)
 *        - Status
 *
 *
 *          Each page:
 *            Transactions
 *            Count (raw requests)
 *            Avg
 *            P90
 *            P95
 *            P99
 *            Max
 *            2xx
 *            3xx
 *            4xx
 *            5xx
 *            Error %
 *
 * 3. Overall HTTP Status Distribution
 *
 * 4. Cache / Backend Breakdown (incl. ETag, Vary, Cache-Control, max-age)
 *
 * 5. Native k6 Metrics
 *
 * SCALABILITY
 * -----------
 * No per-request arrays.
 * No per-request logging.
 * No response-body retention.
 * No per-request report objects.
 *
 * Reporting remains O(phases x URLs), not O(requests).
 *
 * =============================================================================
 */

function statusClass(status) {
    if (status === 0) return 'err';
    if (status < 300) return '2xx';
    if (status < 400) return '3xx';
    if (status < 500) return '4xx';
    return '5xx';
}

// =============================================================================
// 4. URL WEIGHT TABLE
// =============================================================================
//
// Built once during initialization.
// No table construction occurs inside runTest().
// =============================================================================

const URL_TABLE = (function buildUrlTable(urls) {

    const total = urls.reduce(
        (sum, u) => sum + u.weight,
        0
    );

    let cumulative = 0;
    return urls.map((u) => {
        cumulative += u.weight;
        return {
            ...u,
            cumWeight: cumulative / total,
        };
    });

})(CONFIG.urls);


// function to pick a URL based on weights
function pickUrl() {
    const r = Math.random();
    for (let i = 0; i < URL_TABLE.length; i++) {
        if (r <= URL_TABLE[i].cumWeight) {
            return URL_TABLE[i];
        }
    }

    return URL_TABLE[URL_TABLE.length - 1];
}

// 6. DURATION PARSING
// =============================================================================

function parseDurationToSeconds(dur) {
    const re = /(\d+)\s*(h|m|s)/g;
    let match, totalSeconds = 0, found = false;

    while ((match = re.exec(dur)) !== null) {
        found = true;
        const value = parseInt(match[1], 10);
        const unit = match[2];

        if (unit === 'h') totalSeconds += value * 3600;
        else if (unit === 'm') totalSeconds += value * 60;
        else totalSeconds += value;
    }

    if (!found) throw new Error(`Cannot parse duration: ${dur}`);
    return totalSeconds;
}

// =============================================================================
// 7. SCENARIOS
// =============================================================================
//
// One constant-arrival-rate scenario per phase.
//
// Phases are scheduled back-to-back.
//
// The scenario name becomes the automatic `scenario` metric tag.
// =============================================================================

function buildScenarios(config) {
    const scenarios = {};
    let cumulativeStartSeconds = 0;
    config.phases.forEach((phase) => {
        const durationSeconds =
            parseDurationToSeconds(phase.duration);
        const preAllocatedVUs =
            phase.preAllocatedVUs ||
            Math.max(
                10,
                Math.ceil(
                    phase.targetTPS *
                    config.expectedLatencySec
                )
            );

        const maxVUs =
            phase.maxVUs ||
            preAllocatedVUs * 3;

        scenarios[phase.name] = {
            executor: 'constant-arrival-rate',
            rate: phase.targetTPS,
            timeUnit: '1s',
            duration: phase.duration,
            preAllocatedVUs,
            maxVUs,
            startTime: `${cumulativeStartSeconds}s`,
            exec: 'runTest',
        };
        cumulativeStartSeconds += durationSeconds;
    });
    return scenarios;
}

// =============================================================================
// 8. METRIC TAG HELPERS
// =============================================================================
//
// IMPORTANT:
//
// These helpers create the exact phase + URL intersections used by reporting.
//
// NOTE: the custom tag is called "page" (not "url").
// k6 already auto-tags every HTTP metric sample with a built-in system tag
// named "url" (the literal request URL). A user-defined tag sharing that
// name is silently dropped for http_reqs / http_req_duration / etc., so the
// submetrics below would always read back empty if we called it "url".
// "page" avoids the collision.
//
// Example:
//
// http_req_duration{
//   scenario:phase_10000tps,
//   page:homepage_gamma
// }
//
// This is what allows:
//
// Phase 10,000 TPS
//   -> Homepage response time
//   -> FAQ response time
//   -> ContactUs response time
//
// without storing individual requests.
// =============================================================================

function urlTag(name, url) {
    return `${name}{page:${url}}`;
}

function phaseTag(name, phase) {
    return `${name}{scenario:${phase}}`;
}

function phaseUrlTag(name, phase, url) {
    return `${name}{scenario:${phase},page:${url}}`;
}

function phaseUrlClassTag(name, phase, url, cls) {
    return `${name}{scenario:${phase},page:${url},class:${cls}}`;
}

function urlClassTag(name, url, cls) {
    return `${name}{page:${url},class:${cls}}`;
}

function phaseClassTag(name, phase, cls) {
    return `${name}{scenario:${phase},class:${cls}}`;
}

function urlDirectiveTag(name, url, directive) {
    return `${name}{page:${url},directive:${directive}}`;
}

// =============================================================================
// 11. TEST FUNCTION
// =============================================================================
//
// Keep this section extremely small.
//
// This function executes once for every arrival.
//
// There is deliberately:
//
// - no logging
// - no report array
// - no response body parsing
// - no JSON parsing
// - no HTML parsing
// =============================================================================

// Builds the request URL, injecting basic-auth credentials only for pages
// that have opted in via `auth: { username, password }`. Pages without an
// `auth` block are completely unaffected.
function buildRequestUrl(urlConf) {

    if (!urlConf.auth || !urlConf.auth.username) {
        return urlConf.url;
    }

    const user = encodeURIComponent(urlConf.auth.username);
    const pass = encodeURIComponent(urlConf.auth.password || '');

    // k6's http module expects basic-auth credentials embedded directly in
    // the request URL (scheme://user:pass@host/...) alongside `auth: 'basic'`.
    return urlConf.url.replace(
        /^(https?:\/\/)/,
        `$1${user}:${pass}@`
    );
}

export function runTest() {

    const urlConf = pickUrl();

    const id = urlConf.id;

    // One transaction per iteration, regardless of how many HTTP hops the
    // request ends up taking (e.g. redirects). See note on transactionCount.
    transactionCount.add(
        1,
        {
            page: id,
        }
    );

    const requestUrl =
        buildRequestUrl(urlConf);

    const params = {
        tags: {
            page: id,
        },

        responseType: 'none',
    };

    // Only pages configured with `auth` get basic-auth applied.
    if (urlConf.auth && urlConf.auth.username) {
        params.auth = 'basic';
    }

    const res = http.request(
        urlConf.method || 'GET',
        requestUrl,
        urlConf.body || null,
        params
    );

    // Status classification.
    statusClassCount.add(
        1,
        {
            page: id,
            class: statusClass(res.status),
        }
    );

    // Cache/backend classification.
    recordCache(
        res,
        urlConf
    );
}

// =============================================================================
// 12. CASE-INSENSITIVE RESPONSE HEADER LOOKUP
// =============================================================================

function getHeaderCI(headers, name) {

    if (!headers || !name) {
        return undefined;
    }

    const lower =
        name.toLowerCase();

    for (const key in headers) {

        if (key.toLowerCase() === lower) {

            return headers[key];
        }
    }

    return undefined;
}

// =============================================================================
// 13. CACHE / BACKEND RECORDING
// =============================================================================

function recordCache(res, urlConf) {

    const id = urlConf.id;

    const strategy =
        CACHE_STRATEGY[urlConf.provider] ||
        'header';

    // ---------------------------------------------------------------------------
    // No cache layer
    // ---------------------------------------------------------------------------

    if (strategy === 'not-applicable') {

        cacheNotApplicableCount.add(
            1,
            {
                page: id,
            }
        );

        return;
    }

    // ---------------------------------------------------------------------------
    // GCP CDN / GCLB -- no explicit hit/miss header, so use Age as the
    // primary signal: Age > 0 means it was served from cache, Age missing
    // or 0 means it came from the origin.
    // ---------------------------------------------------------------------------

    const cfg =
        CONFIG.cacheHeaders;

    const headers =
        res.headers;

    if (strategy === 'gcp-age') {

        const ageValueRaw = getHeaderCI(headers, cfg.ageHeader);
        const ageNum = ageValueRaw !== undefined ? parseFloat(ageValueRaw) : NaN;

        if (!isNaN(ageNum)) {

            agePresentCount.add(1, { page: id });
            cacheAge.add(ageNum, { page: id });

            if (ageNum > 0) {
                cacheHitCount.add(1, { page: id });
            } else {
                cacheMissCount.add(1, { page: id });
            }

        } else {
            // No Age header at all -- can't infer hit/miss for this response.
            cacheUnknownCount.add(1, { page: id });
        }

        // Fall through to the shared ETag / Vary / Cache-Control / backend
        // capture below, since those are independent of the strategy.

    } else if (strategy === 'not-observable') {

        cacheNotObservableCount.add(
            1,
            {
                page: id,
            }
        );

        return;

    } else {

        // ---------------------------------------------------------------------------
        // Header-based CDN classification (explicit X-Cache / CF-Cache-Status /
        // Akamai / Varnish style headers)
        // ---------------------------------------------------------------------------

        let classified = false;

        for (
            const headerName
            of cfg.statusHeaders
        ) {

            const value =
                getHeaderCI(
                    headers,
                    headerName
                );

            if (value === undefined) {
                continue;
            }

            const v = String(value).toLowerCase();

            if (v.includes('hit')) {
                cacheHitCount.add(1, { page: id });
                classified = true;
                break;
            }

            if (v.includes('miss')) {
                cacheMissCount.add(1, { page: id });
                classified = true;
                break;
            }
        }

        if (!classified) cacheUnknownCount.add(1, { page: id });

        // ---------------------------------------------------------------------------
        // Age (informational here -- 'header' strategy sites use an explicit
        // status header as the primary signal, not Age)
        // ---------------------------------------------------------------------------

        const ageValue = getHeaderCI(headers, cfg.ageHeader);

        if (ageValue !== undefined) {
            const ageNum = parseFloat(ageValue);
            if (!isNaN(ageNum)) {
                agePresentCount.add(1, { page: id });
                cacheAge.add(ageNum, { page: id });
            }
        }
    }

    // ---------------------------------------------------------------------------
    // ETag presence
    //
    // ETag alone doesn't say hit/miss, but its presence (and whether it's
    // weak "W/..." or strong) is useful for validating conditional-caching
    // / revalidation behavior on a page.
    // ---------------------------------------------------------------------------

    const etagValue = getHeaderCI(headers, cfg.etagHeader);

    if (etagValue !== undefined && String(etagValue).length > 0) {
        etagPresentCount.add(1, { page: id });
    }

    // ---------------------------------------------------------------------------
    // Vary presence
    // ---------------------------------------------------------------------------

    const varyValue = getHeaderCI(headers, cfg.varyHeader);

    if (varyValue !== undefined && String(varyValue).length > 0) {
        varyPresentCount.add(1, { page: id });
    }

    // ---------------------------------------------------------------------------
    // Cache-Control directives + max-age
    // ---------------------------------------------------------------------------

    const cacheControlValue = getHeaderCI(headers, cfg.cacheControlHeader);

    if (cacheControlValue !== undefined) {

        cacheControlPresentCount.add(1, { page: id });

        const directives =
            String(cacheControlValue)
                .toLowerCase()
                .split(',')
                .map((d) => d.trim())
                .filter(Boolean);

        directives.forEach((directive) => {

            const [dirName, dirVal] = directive.split('=');

            if (CACHE_CONTROL_DIRECTIVES.includes(dirName)) {
                cacheControlDirectiveCount.add(1, { page: id, directive: dirName });
            }

            if (dirName === 'max-age' && dirVal !== undefined) {
                const maxAgeNum = parseFloat(dirVal);
                if (!isNaN(maxAgeNum)) cacheControlMaxAge.add(maxAgeNum, { page: id });
            }
        });
    }

    // ---------------------------------------------------------------------------
    // Backend response indicator
    // ---------------------------------------------------------------------------

    const backendValue = getHeaderCI(headers, cfg.backendHeader);

    if (backendValue !== undefined && String(backendValue).toLowerCase() === 'true') {
        backendResponseCount.add(1, { page: id });
    }

    // ---------------------------------------------------------------------------
    // Cache response indicator
    // ---------------------------------------------------------------------------

    const cacheRespValue = getHeaderCI(headers, cfg.cacheResponseHeader);

    if (cacheRespValue !== undefined && String(cacheRespValue).toLowerCase() === 'true') {
        cacheResponseCount.add(1, { page: id });
    }
}

// =============================================================================
// 14. REPORTING HELPERS
// =============================================================================

function metric(data, name) {

    return (
        data.metrics &&
        data.metrics[name]
    ) || null;
}

function trendStat(m, stat, fallback = 0) {
    return (m && m.values && m.values[stat] !== undefined) ? m.values[stat] : fallback;
}

function counterCount(m, fallback = 0) {
    return (m && m.values && m.values.count !== undefined) ? m.values.count : fallback;
}

function rateValue(m, fallback = 0) {
    return (m && m.values && m.values.rate !== undefined) ? m.values.rate : fallback;
}

function gaugeMax(m, fallback = 0) {
    return (m && m.values && m.values.max !== undefined) ? m.values.max : fallback;
}

function pct(part, whole) {
    return whole > 0 ? (part / whole) * 100 : 0;
}

function fmt(n, decimals = 2) {
    if (n === null || n === undefined || isNaN(n)) return 'n/a';
    return Number(n).toFixed(decimals);
}

function esc(str) {
    return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

function fmtBytes(n) {
    if (n === null || n === undefined || isNaN(n)) return 'n/a';
    const num = Number(n);
    if (num < 1024) return `${num.toFixed(0)} B`;
    if (num < 1024 * 1024) return `${(num / 1024).toFixed(2)} KB`;
    return `${(num / (1024 * 1024)).toFixed(2)} MB`;
}

// =============================================================================
// 15. BUILD PAGE METRICS FOR A SPECIFIC PHASE
// =============================================================================
//
// THIS is the main reporting change.
//
// Every page is now explicitly scoped to:
//
//     scenario = phase
//     url      = page
//
// Therefore:
//
// Phase 1000 TPS
//   Homepage response metrics
//   FAQ response metrics
//   ContactUs response metrics
//
// Phase 5000 TPS
//   Homepage response metrics
//   FAQ response metrics
//   ContactUs response metrics
//
// etc.
// =============================================================================

function buildPhasePageRow(
    data,
    phase,
    urlConf
) {

    const phaseName =
        phase.name;

    const id =
        urlConf.id;

    // ---------------------------------------------------------------------------
    // Phase + URL transaction count (logical requests, excl. redirect hops)
    // ---------------------------------------------------------------------------

    const transactions =
        counterCount(
            metric(
                data,
                phaseUrlTag(
                    'transaction_count',
                    phaseName,
                    id
                )
            )
        );

    // ---------------------------------------------------------------------------
    // Phase + URL request count (raw HTTP requests, incl. redirect hops)
    // ---------------------------------------------------------------------------

    const reqs =
        counterCount(
            metric(
                data,
                phaseUrlTag(
                    'http_reqs',
                    phaseName,
                    id
                )
            )
        );

    // ---------------------------------------------------------------------------
    // Phase + URL response time
    // ---------------------------------------------------------------------------

    const durationM =
        metric(
            data,
            phaseUrlTag(
                'http_req_duration',
                phaseName,
                id
            )
        );

    // ---------------------------------------------------------------------------
    // Phase + URL status classes
    // ---------------------------------------------------------------------------

    const classCounts = {};

    STATUS_CLASSES.forEach((cls) => {

        classCounts[cls] =
            counterCount(
                metric(
                    data,
                    phaseUrlClassTag(
                        'status_class_count',
                        phaseName,
                        id,
                        cls
                    )
                )
            );
    });

    const failed =
        classCounts['4xx'] +
        classCounts['5xx'] +
        classCounts['err'];

    const passed =
        Math.max(
            0,
            reqs - failed
        );

    const errorPct =
        pct(
            failed,
            reqs
        );

    return {

        id,

        name:
            urlConf.name || id,

        url:
            urlConf.url,

        authEnabled:
            !!(urlConf.auth && urlConf.auth.username),

        transactions,

        count:
            reqs,

        avg:
            trendStat(
                durationM,
                'avg'
            ),

        min:
            trendStat(
                durationM,
                'min'
            ),

        p90:
            trendStat(
                durationM,
                'p(90)'
            ),

        p95:
            trendStat(
                durationM,
                'p(95)'
            ),

        p99:
            trendStat(
                durationM,
                'p(99)'
            ),

        max:
            trendStat(
                durationM,
                'max'
            ),

        passed,

        failed,

        errorPct,

        classCounts,

        status:
            errorPct <=
                CONFIG.reporting.maxErrorRatePct
                ? 'PASS'
                : 'FAIL',
    };
}

// =============================================================================
// 16. BUILD PHASE ROW
// =============================================================================
//
// Each phase contains its own pages[] collection.
//
// This gives the report hierarchy:
//
// phase
//   -> TPS
//   -> phase status
//   -> pages
//        -> Homepage
//        -> FAQ
//        -> ContactUs
// =============================================================================

function buildPhaseRow(
    data,
    phase
) {

    const name =
        phase.name;

    // ---------------------------------------------------------------------------
    // Phase transaction count (logical) vs request count (raw, incl. redirects)
    // ---------------------------------------------------------------------------

    const transactions =
        counterCount(
            metric(
                data,
                phaseTag(
                    'transaction_count',
                    name
                )
            )
        );

    const reqs =
        counterCount(
            metric(
                data,
                phaseTag(
                    'http_reqs',
                    name
                )
            )
        );

    // ---------------------------------------------------------------------------
    // Phase response time
    //
    // Kept for overall phase-level context.
    // Page-level values are shown below it.
    // ---------------------------------------------------------------------------

    const durationM =
        metric(
            data,
            phaseTag(
                'http_req_duration',
                name
            )
        );

    // ---------------------------------------------------------------------------
    // Phase status classes
    // ---------------------------------------------------------------------------

    const classCounts = {};

    STATUS_CLASSES.forEach((cls) => {

        classCounts[cls] =
            counterCount(
                metric(
                    data,
                    phaseClassTag(
                        'status_class_count',
                        name,
                        cls
                    )
                )
            );
    });

    const failed =
        classCounts['4xx'] +
        classCounts['5xx'] +
        classCounts['err'];

    const passed =
        Math.max(
            0,
            reqs - failed
        );

    const errorPct =
        pct(
            failed,
            reqs
        );

    // ---------------------------------------------------------------------------
    // Phase duration / actual TPS
    //
    // Actual TPS is based on transactions (logical requests), not raw
    // http_reqs, so it isn't inflated by redirect hops.
    // ---------------------------------------------------------------------------

    const durationSeconds =
        parseDurationToSeconds(
            phase.duration
        );

    const actualTps =
        durationSeconds > 0
            ? transactions / durationSeconds
            : 0;

    // ---------------------------------------------------------------------------
    // Dropped iterations
    //
    // Tracked for visibility only. Per requirements, dropped iterations do
    // NOT cause a phase or the overall run to fail -- see `status` below,
    // which intentionally omits dropOk from the pass/fail decision.
    // ---------------------------------------------------------------------------

    const dropped =
        counterCount(
            metric(
                data,
                phaseTag(
                    'dropped_iterations',
                    name
                )
            )
        );

    // ---------------------------------------------------------------------------
    // TPS achievement
    // ---------------------------------------------------------------------------

    const tpsAchievementPct =
        pct(
            actualTps,
            phase.targetTPS
        );

    const tpsOk =
        tpsAchievementPct >=
        CONFIG.reporting.minTpsAchievementPct;

    const errOk =
        errorPct <=
        CONFIG.reporting.maxErrorRatePct;

    // Informational only -- intentionally NOT part of `status` below.
    const dropOk =
        dropped === 0;

    // ---------------------------------------------------------------------------
    // PAGE METRICS INSIDE THIS PHASE
    // ---------------------------------------------------------------------------

    const pages =
        CONFIG.urls.map(
            (urlConf) =>
                buildPhasePageRow(
                    data,
                    phase,
                    urlConf
                )
        );

    const networkMetrics = buildPhaseNetworkMetrics(data, name);

    return {
        phase: name,
        targetTps: phase.targetTPS,
        transactions,
        actualTps,
        tpsAchievementPct,
        duration: phase.duration,
        avg: trendStat(durationM, 'avg'),
        min: trendStat(durationM, 'min'),
        p90: trendStat(durationM, 'p(90)'),
        p95: trendStat(durationM, 'p(95)'),
        p99: trendStat(durationM, 'p(99)'),
        max: trendStat(durationM, 'max'),
        count: reqs,
        passed,
        failed,
        errorPct,
        classCounts,
        dropped,
        tpsOk,
        errOk,
        dropOk,
        // Dropped iterations deliberately excluded from pass/fail.
        status: tpsOk && errOk ? 'PASS' : 'FAIL',
        pages,
        networkMetrics,
    };
}

// =============================================================================
// 16b. NATIVE NETWORK METRICS FOR A PHASE
// =============================================================================
//
// Pulls back out the native k6 metrics that were already registered as
// phase-scoped submetrics in buildThresholds(). No new metric objects are
// created here -- this only reads data k6 was already collecting.
// =============================================================================

function buildPhaseNetworkMetrics(
    data,
    phaseName
) {

    const trends =
        NETWORK_TREND_METRICS.map((m) => {

            const met =
                metric(
                    data,
                    phaseTag(
                        m.key,
                        phaseName
                    )
                );

            return {

                key:
                    m.key,

                label:
                    m.label,

                avg:
                    trendStat(met, 'avg'),

                min:
                    trendStat(met, 'min'),

                p90:
                    trendStat(met, 'p(90)'),

                p95:
                    trendStat(met, 'p(95)'),

                p99:
                    trendStat(met, 'p(99)'),

                max:
                    trendStat(met, 'max'),
            };
        });

    const dataSentBytes =
        counterCount(
            metric(
                data,
                phaseTag('data_sent', phaseName)
            )
        );

    const dataReceivedBytes =
        counterCount(
            metric(
                data,
                phaseTag('data_received', phaseName)
            )
        );

    const iterDurM =
        metric(
            data,
            phaseTag('iteration_duration', phaseName)
        );

    const iterationDuration = {

        avg:
            trendStat(iterDurM, 'avg'),

        min:
            trendStat(iterDurM, 'min'),

        p90:
            trendStat(iterDurM, 'p(90)'),

        p95:
            trendStat(iterDurM, 'p(95)'),

        p99:
            trendStat(iterDurM, 'p(99)'),

        max:
            trendStat(iterDurM, 'max'),
    };

    const httpReqFailedM =
        metric(
            data,
            phaseTag('http_req_failed', phaseName)
        );

    const httpReqFailedPct =
        rateValue(httpReqFailedM) * 100;

    return {

        trends,

        dataSentBytes,

        dataReceivedBytes,

        iterationDuration,

        httpReqFailedPct,
    };
}

// =============================================================================
// 17. CACHE REPORT
// =============================================================================

function buildCacheRow(data, urlConf) {
    const id = urlConf.id;
    const strategy = CACHE_STRATEGY[urlConf.provider] || 'header';

    if (strategy === 'not-applicable') {
        return { id, name: urlConf.name || id, url: urlConf.url, strategy, note: 'No CDN in front of this URL -- not applicable.' };
    }

    if (strategy === 'not-observable') {
        return { id, name: urlConf.name || id, url: urlConf.url, strategy, note: 'GCP Cloud CDN does not expose cache status to the client. Check cache_result in Cloud Monitoring / Cloud Logging for the test time window.' };
    }

    const hit = counterCount(metric(data, urlTag('cache_hit_count', id)));
    const miss = counterCount(metric(data, urlTag('cache_miss_count', id)));
    const unknown = counterCount(metric(data, urlTag('cache_unknown_count', id)));
    const backend = counterCount(metric(data, urlTag('backend_response_count', id)));
    const cacheResp = counterCount(metric(data, urlTag('cache_response_count', id)));
    const classifiable = hit + miss;

    // Basis for presence percentages: total transactions for this page
    // across the whole run (not just classifiable hit/miss ones).
    const transactions = counterCount(metric(data, urlTag('transaction_count', id)));

    // ---------------------------------------------------------------------------
    // Age: YES/NO + the observed value range.
    // ---------------------------------------------------------------------------

    const agePresent = counterCount(metric(data, urlTag('age_present_count', id)));
    const ageM = metric(data, urlTag('cache_age', id));

    const age = {
        present: agePresent > 0,
        presentPct: pct(agePresent, transactions),
        avg: trendStat(ageM, 'avg'),
        min: trendStat(ageM, 'min'),
        max: trendStat(ageM, 'max'),
        p90: trendStat(ageM, 'p(90)'),
    };

    // ---------------------------------------------------------------------------
    // Cache-Control: YES/NO + reconstructed value.
    //
    // We can't store the raw header string itself (k6 thresholds/submetrics
    // have to be declared ahead of time, so an arbitrary runtime string
    // can't be captured as one series). Instead we reconstruct the value
    // from the bounded set of directives + max-age we DO track: any
    // directive seen on every single Cache-Control response is treated as
    // part of the "typical" value. In practice Cache-Control is static per
    // page/origin config (unlike Age, which legitimately changes per
    // request), so this reproduces the real value. If directives are
    // inconsistent across responses, that's flagged instead of guessing.
    // ---------------------------------------------------------------------------

    const cacheControlPresent = counterCount(metric(data, urlTag('cache_control_present_count', id)));
    const avgMaxAge = trendStat(metric(data, urlTag('cache_control_max_age', id)), 'avg');

    const directiveCounts = {};
    CACHE_CONTROL_DIRECTIVES.forEach((dir) => {
        directiveCounts[dir] = counterCount(metric(data, urlDirectiveTag('cache_control_directive_count', id, dir)));
    });

    const directiveSummary =
        CACHE_CONTROL_DIRECTIVES
            .map((dir) => (directiveCounts[dir] > 0 ? `${dir}:${directiveCounts[dir]}` : null))
            .filter(Boolean)
            .join(', ') || 'n/a';

    let cacheControlTypicalValue = 'n/a';
    let cacheControlConsistent = true;

    if (cacheControlPresent > 0) {

        const parts = [];

        CACHE_CONTROL_DIRECTIVES.forEach((dir) => {

            const count = directiveCounts[dir];

            if (count === 0) {
                return;
            }

            if (count < cacheControlPresent) {
                cacheControlConsistent = false;
            }

            if (dir === 'max-age') {
                parts.push(`max-age=${fmt(avgMaxAge, 0)}`);
            } else {
                parts.push(dir);
            }
        });

        cacheControlTypicalValue =
            parts.length > 0
                ? parts.join(', ')
                : 'present (no tracked directives matched)';
    }

    const cacheControl = {
        present: cacheControlPresent > 0,
        presentPct: pct(cacheControlPresent, transactions),
        typicalValue: cacheControlTypicalValue,
        consistent: cacheControlConsistent,
        directiveCounts,
        directiveSummary,
        sampleCount: cacheControlPresent,
    };

    const etagPresent = counterCount(metric(data, urlTag('etag_present_count', id)));
    const varyPresent = counterCount(metric(data, urlTag('vary_present_count', id)));

    return {
        id,
        name: urlConf.name || id,
        url: urlConf.url,
        strategy,
        hit,
        miss,
        unknown,
        hitPct: pct(hit, classifiable),
        backend,
        cacheResp,
        transactions,
        age,
        cacheControl,
        etagPresent,
        etagPct: pct(etagPresent, transactions),
        varyPresent,
        varyPct: pct(varyPresent, transactions),
    };
}

// =============================================================================
// 19. HANDLE SUMMARY
// =============================================================================
//
// Runs once after the test.
//
// All data here is already aggregated by k6.
// =============================================================================

export function handleSummary(data) {

    const durationMs =
        data.state &&
            data.state.testRunDurationMs
            ? data.state.testRunDurationMs
            : 0;

    const durationSec =
        durationMs / 1000;

    const endTime =
        new Date();

    const startTime =
        new Date(
            endTime.getTime() -
            durationMs
        );

    // ===========================================================================
    // OVERALL REQUESTS / TRANSACTIONS
    // ===========================================================================

    const overallReqs =
        counterCount(
            metric(
                data,
                'http_reqs'
            )
        );

    // Sum the bounded per-URL transaction counts -- same aggregation
    // pattern used for overallClassCounts below, so this stays
    // O(phases x URLs), not O(requests).
    const overallTransactions =
        CONFIG.urls.reduce(
            (sum, u) =>
                sum +
                counterCount(
                    metric(
                        data,
                        urlTag('transaction_count', u.id)
                    )
                ),
            0
        );

    const overallDurationM =
        metric(
            data,
            'http_req_duration'
        );

    // ===========================================================================
    // OVERALL STATUS CLASSES
    // ===========================================================================
    //
    // Sum the bounded URL-level class metrics.
    // This does NOT scan individual requests.
    // ===========================================================================

    const overallClassCounts = {};

    STATUS_CLASSES.forEach((cls) => {

        overallClassCounts[cls] =
            CONFIG.urls.reduce(
                (sum, u) => {

                    return sum +
                        counterCount(
                            metric(
                                data,
                                urlClassTag(
                                    'status_class_count',
                                    u.id,
                                    cls
                                )
                            )
                        );
                },
                0
            );
    });

    const overallFailed =
        overallClassCounts['4xx'] +
        overallClassCounts['5xx'] +
        overallClassCounts['err'];

    const overallPassed =
        Math.max(
            0,
            overallReqs -
            overallFailed
        );

    const overallErrorPct =
        pct(
            overallFailed,
            overallReqs
        );

    const overallDropped =
        counterCount(
            metric(
                data,
                'dropped_iterations'
            )
        );

    // ===========================================================================
    // TARGET TPS
    // ===========================================================================

    const overallTargetTps =
        CONFIG.phases.reduce(
            (max, p) =>
                Math.max(
                    max,
                    p.targetTPS
                ),
            0
        );

    // ===========================================================================
    // OVERALL ACTUAL TPS
    // ===========================================================================
    //
    // This is informational only because multiple phases can have different
    // target rates. Based on transactions (logical requests), not raw
    // http_reqs, so redirect hops don't inflate it.
    //
    // The authoritative TPS assessment is in the phase table.
    // ===========================================================================

    const overallActualTps =
        durationSec > 0
            ? overallTransactions / durationSec
            : 0;

    // ===========================================================================
    // OVERALL ACCEPTANCE
    // ===========================================================================
    //
    // Dropped iterations are intentionally excluded from the pass/fail
    // decision -- dropOk is still computed and shown, but not AND'ed in.
    // ===========================================================================

    const errOk =
        overallErrorPct <=
        CONFIG.reporting.maxErrorRatePct;

    const dropOk =
        overallDropped === 0;

    // For multi-phase tests, phase-level TPS achievement is authoritative.
    const allPhasesTpsOk =
        CONFIG.phases.every((phase) => {

            const phaseTransactions =
                counterCount(
                    metric(
                        data,
                        phaseTag(
                            'transaction_count',
                            phase.name
                        )
                    )
                );

            const phaseDuration =
                parseDurationToSeconds(
                    phase.duration
                );

            const phaseActualTps =
                phaseDuration > 0
                    ? phaseTransactions /
                    phaseDuration
                    : 0;

            const achievement =
                pct(
                    phaseActualTps,
                    phase.targetTPS
                );

            return achievement >=
                CONFIG.reporting.minTpsAchievementPct;
        });

    const overallStatus =
        allPhasesTpsOk &&
            errOk
            ? 'PASS'
            : 'FAIL';

    // ===========================================================================
    // FINAL SUMMARY OBJECT
    // ===========================================================================

    const summary = {

        meta: {

            testType:
                CONFIG.testType,

            startTime:
                startTime.toISOString(),

            endTime:
                endTime.toISOString(),

            durationSec,

            pageCount:
                CONFIG.urls.length,

            phaseCount:
                CONFIG.phases.length,

            errorRateFailThresholdPct:
                CONFIG.reporting.maxErrorRatePct,

            tpsAchievementFailThresholdPct:
                CONFIG.reporting.minTpsAchievementPct,
        },

        // -------------------------------------------------------------------------
        // OVERALL
        // -------------------------------------------------------------------------

        overall: {

            count:
                overallReqs,

            transactions:
                overallTransactions,

            passed:
                overallPassed,

            failed:
                overallFailed,

            errorPct:
                overallErrorPct,

            targetTps:
                overallTargetTps,

            actualTps:
                overallActualTps,

            dropped:
                overallDropped,

            avg:
                trendStat(
                    overallDurationM,
                    'avg'
                ),

            min:
                trendStat(
                    overallDurationM,
                    'min'
                ),

            p90:
                trendStat(
                    overallDurationM,
                    'p(90)'
                ),

            p95:
                trendStat(
                    overallDurationM,
                    'p(95)'
                ),

            p99:
                trendStat(
                    overallDurationM,
                    'p(99)'
                ),

            max:
                trendStat(
                    overallDurationM,
                    'max'
                ),

            // vus_max is a Gauge, so use max rather than Counter.count.
            vusMax:
                gaugeMax(
                    metric(
                        data,
                        'vus_max'
                    )
                ),

            classCounts:
                overallClassCounts,

            status:
                overallStatus,

            tpsOk:
                allPhasesTpsOk,

            errOk,

            // Informational only -- not part of `status`.
            dropOk,
        },

        // -------------------------------------------------------------------------
        // PHASES
        //
        // Each phase contains pages[].
        // -------------------------------------------------------------------------

        phases:
            CONFIG.phases.map(
                (phase) =>
                    buildPhaseRow(
                        data,
                        phase
                    )
            ),

        // -------------------------------------------------------------------------
        // CACHE
        // -------------------------------------------------------------------------

        cache:
            CONFIG.urls.map(
                (url) =>
                    buildCacheRow(
                        data,
                        url
                    )
            ),

        // -------------------------------------------------------------------------
        // Native aggregate metrics.
        //
        // Retained for future dashboarding.
        // -------------------------------------------------------------------------

        nativeMetrics:
            data.metrics,
    };

    return {

        'k6-report.html':
            buildHtmlReport(
                summary
            ),

        'k6-summary.json':
            JSON.stringify(
                summary,
                null,
                2
            ),

        stdout:
            textSummary(
                data,
                {
                    indent: ' ',
                    enableColors: true,
                }
            ) +
            `\nTest complete. Overall status: ${summary.overall.status}. ` +
            `Transactions: ${summary.overall.transactions}, Requests: ${summary.overall.count}, ` +
            `Dropped (info only): ${summary.overall.dropped}. ` +
            `See k6-report.html and k6-summary.json.\n`,
    };
}
