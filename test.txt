 * =============================================================================
 * High-Throughput k6 Performance Testing Framework
 * =============================================================================
 *
 * REPORT STRUCTURE
 * ----------------
 *
 * 1. Overall Summary
 *
 * 2. Phase-wise Breakdown
 *      Phase
 *        - Target TPS
 *        - Actual TPS
 *        - Requests
 *        - Dropped iterations
 *        - Error %
 *        - Status

 *

 * 4. Overall HTTP Status Distribution

 * SCALABILITY
 * -----------
 * No per-request arrays.
 * No per-request logging.
 * No response-body retention.
 * No per-request report objects.
 *
 * Reporting remains O(phases x URLs), not O(requests).

 * =============================================================================
 */

function statusClass(status) {
    if (status === 0) return 'err';
    if (status < 300) return '2xx';
    if (status < 400) return '3xx';
    if (status < 500) return '4xx';
    return '5xx';
}

// Treat 4xx / 5xx as failed HTTP requests.

    http.expectedStatuses({
        min: 200,
        max: 399,
    })
);

// =============================================================================
// 4. URL WEIGHT TABLE
// =============================================================================
//
// Built once during initialization.
// No table construction occurs inside runTest().
// =============================================================================

const URL_TABLE = (function buildUrlTable(urls) {

    const total = urls.reduce(
        (sum, u) => sum + u.weight,
        0
    );

    let cumulative = 0;
    return urls.map((u) => {
        cumulative += u.weight;
        return {
            ...u,
            cumWeight: cumulative / total,
        };
    });

})(CONFIG.urls);


// function to pick a URL based on weights
function pickUrl() {
    const r = Math.random();
    for (let i = 0; i < URL_TABLE.length; i++) {
        if (r <= URL_TABLE[i].cumWeight) {
            return URL_TABLE[i];
        }
    }

    return URL_TABLE[URL_TABLE.length - 1];
}

// =============================================================================
// 6. DURATION PARSING
// =============================================================================

function parseDurationToSeconds(dur) {
    const re = /(\d+)\s*(h|m|s)/g;
    let match, totalSeconds = 0, found = false;

    while ((match = re.exec(dur)) !== null) {
        found = true;
        const value = parseInt(match[1], 10);
        const unit = match[2];

        if (unit === 'h') totalSeconds += value * 3600;
        else if (unit === 'm') totalSeconds += value * 60;
        else totalSeconds += value;
    }

    if (!found) throw new Error(`Cannot parse duration: ${dur}`);
    return totalSeconds;
}

// =============================================================================
// 7. SCENARIOS
// =============================================================================
//
// One constant-arrival-rate scenario per phase.
//
// Phases are scheduled back-to-back.
//
// The scenario name becomes the automatic `scenario` metric tag.
// =============================================================================

function buildScenarios(config) {
    const scenarios = {};
    let cumulativeStartSeconds = 0;
    config.phases.forEach((phase) => {
        const durationSeconds =
            parseDurationToSeconds(phase.duration);
        const preAllocatedVUs =
            phase.preAllocatedVUs ||
            Math.max(
                10,
                Math.ceil(
                    phase.targetTPS *
                    config.expectedLatencySec
                )
            );

        const maxVUs =
            phase.maxVUs ||
            preAllocatedVUs * 3;

        scenarios[phase.name] = {
            executor: 'constant-arrival-rate',
            rate: phase.targetTPS,
            timeUnit: '1s',
            duration: phase.duration,
            preAllocatedVUs,
            maxVUs,
            startTime: `${cumulativeStartSeconds}s`,
            exec: 'runTest',
        };
        cumulativeStartSeconds += durationSeconds;
    });
    return scenarios;
}

// =============================================================================
// 8. METRIC TAG HELPERS
// =============================================================================
//
// IMPORTANT:
//
// These helpers create the exact phase + URL intersections used by reporting.
//
// NOTE: the custom tag is called "page" (not "url").
// k6 already auto-tags every HTTP metric sample with a built-in system tag
// named "url" (the literal request URL). A user-defined tag sharing that
// name is silently dropped for http_reqs / http_req_duration / etc., so the
// submetrics below would always read back empty if we called it "url".
// "page" avoids the collision.
//
// Example:
//
// http_req_duration{
//   scenario:phase_10000tps,
//   page:homepage_gamma
// }
//
// This is what allows:
//
// Phase 10,000 TPS
//   -> Homepage response time
//   -> FAQ response time
//   -> ContactUs response time
//
// without storing individual requests.
// =============================================================================

function urlTag(name, url) {
    return `${name}{page:${url}}`;
}

function phaseTag(name, phase) {
    return `${name}{scenario:${phase}}`;
}

function phaseUrlTag(name, phase, url) {
    return `${name}{scenario:${phase},page:${url}}`;
}

function phaseUrlClassTag(name, phase, url, cls) {
    return `${name}{scenario:${phase},page:${url},class:${cls}}`;
}

function urlClassTag(name, url, cls) {
    return `${name}{page:${url},class:${cls}}`;
}

function phaseClassTag(name, phase, cls) {
    return `${name}{scenario:${phase},class:${cls}}`;
}

// =============================================================================
// 9. THRESHOLDS
// =============================================================================
//
// Thresholds are deliberately always-true.
//
// Their purpose here is to register the bounded tagged metric series so they
// can be retrieved from handleSummary().
//
// Number of series:
//
//     phases x URLs x metrics
//
// NOT:
//
//     requests
//
// Therefore 10K TPS / millions of requests does not cause the metric
// cardinality to grow with request volume.
// =============================================================================

function buildThresholds(config) {

    const th = {};

    const urls =
        config.urls.map((u) => u.id);

    const phases =
        config.phases.map((p) => p.name);

    // ---------------------------------------------------------------------------
    // URL-level metrics
    // ---------------------------------------------------------------------------

    urls.forEach((url) => {

        th[urlTag('http_reqs', url)] =
            ['count>=0'];

        th[urlTag('http_req_duration', url)] =
            ['p(99)>=0'];

        th[urlTag('http_req_failed', url)] =
            ['rate>=0'];

        th[urlTag('cache_hit_count', url)] =
            ['count>=0'];

        th[urlTag('cache_miss_count', url)] =
            ['count>=0'];

        th[urlTag('cache_unknown_count', url)] =
            ['count>=0'];

        th[urlTag('cache_not_observable_count', url)] =
            ['count>=0'];

        th[urlTag('cache_not_applicable_count', url)] =
            ['count>=0'];

        th[urlTag('cache_age', url)] =
            ['avg>=0'];

        th[urlTag('backend_response_count', url)] =
            ['count>=0'];

        th[urlTag('cache_response_count', url)] =
            ['count>=0'];

        STATUS_CLASSES.forEach((cls) => {

            th[
                urlClassTag(
                    'status_class_count',
                    url,
                    cls
                )
            ] = ['count>=0'];
        });
    });

    // ---------------------------------------------------------------------------
    // Phase-level metrics
    // ---------------------------------------------------------------------------

    phases.forEach((phase) => {

        th[
            phaseTag(
                'http_reqs',
                phase
            )
        ] = ['count>=0'];

        th[
            phaseTag(
                'http_req_duration',
                phase
            )
        ] = ['p(99)>=0'];

        th[
            phaseTag(
                'http_req_failed',
                phase
            )
        ] = ['rate>=0'];

        th[
            phaseTag(
                'iterations',
                phase
            )
        ] = ['count>=0'];

        th[
            phaseTag(
                'dropped_iterations',
                phase
            )
        ] = ['count>=0'];

        // -------------------------------------------------------------------------
        // Native network-timing metrics (phase-scoped only, not phase+page).
        //
        // These are metrics k6 already emits for every request -- reusing them
        // here just registers the bounded per-phase submetric series.
        // -------------------------------------------------------------------------

        NETWORK_TREND_METRICS.forEach((m) => {

            th[
                phaseTag(
                    m.key,
                    phase
                )
            ] = ['p(99)>=0'];
        });

        th[
            phaseTag(
                'data_sent',
                phase
            )
        ] = ['count>=0'];

        th[
            phaseTag(
                'data_received',
                phase
            )
        ] = ['count>=0'];

        th[
            phaseTag(
                'iteration_duration',
                phase
            )
        ] = ['p(99)>=0'];

        STATUS_CLASSES.forEach((cls) => {

            th[
                phaseClassTag(
                    'status_class_count',
                    phase,
                    cls
                )
            ] = ['count>=0'];
        });

        // -------------------------------------------------------------------------
        // PHASE + URL metrics
        //
        // This is the important new section.
        // -------------------------------------------------------------------------

        urls.forEach((url) => {

            th[
                phaseUrlTag(
                    'http_reqs',
                    phase,
                    url
                )
            ] = ['count>=0'];

            th[
                phaseUrlTag(
                    'http_req_duration',
                    phase,
                    url
                )
            ] = ['p(99)>=0'];

            th[
                phaseUrlTag(
                    'http_req_failed',
                    phase,
                    url
                )
            ] = ['rate>=0'];

            STATUS_CLASSES.forEach((cls) => {

                th[
                    phaseUrlClassTag(
                        'status_class_count',
                        phase,
                        url,
                        cls
                    )
                ] = ['count>=0'];
            });
        });
    });

    return th;
}

// =============================================================================
// 11. TEST FUNCTION
// =============================================================================
//
// Keep this section extremely small.
//
// This function executes once for every arrival.
//
// There is deliberately:
//
// - no logging
// - no report array
// - no response body parsing
// - no JSON parsing
// - no HTML parsing
// =============================================================================

export function runTest() {

    const urlConf = pickUrl();

    const id = urlConf.id;

    const res = http.request(
        urlConf.method || 'GET',
        urlConf.url,
        urlConf.body || null,
        {
            tags: {
                page: id,
            },

            responseType: 'none',
        }
    );

    // Status classification.
    statusClassCount.add(
        1,
        {
            page: id,
            class: statusClass(res.status),
        }
    );

    // Cache/backend classification.
    recordCache(
        res,
        urlConf
    );
}

// =============================================================================
// 12. CASE-INSENSITIVE RESPONSE HEADER LOOKUP
// =============================================================================

function getHeaderCI(headers, name) {

    if (!headers || !name) {
        return undefined;
    }

    const lower =
        name.toLowerCase();

    for (const key in headers) {

        if (key.toLowerCase() === lower) {

            return headers[key];
        }
    }

    return undefined;
}

// =============================================================================
// 13. CACHE / BACKEND RECORDING
// =============================================================================

function recordCache(res, urlConf) {

    const id = urlConf.id;

    const strategy =
        CACHE_STRATEGY[urlConf.provider] ||
        'header';

    // ---------------------------------------------------------------------------
    // No cache layer
    // ---------------------------------------------------------------------------

    if (strategy === 'not-applicable') {

        cacheNotApplicableCount.add(
            1,
            {
                page: id,
            }
        );

        return;
    }

    // ---------------------------------------------------------------------------
    // GCP CDN / GCLB
    //
    // Cache result is not inferred from client headers.
    // ---------------------------------------------------------------------------

    if (strategy === 'not-observable') {

        cacheNotObservableCount.add(
            1,
            {
                page: id,
            }
        );

        return;
    }

    // ---------------------------------------------------------------------------
    // Header-based CDN classification
    // ---------------------------------------------------------------------------

    const cfg =
        CONFIG.cacheHeaders;

    const headers =
        res.headers;

    let classified = false;

    for (
        const headerName
        of cfg.statusHeaders
    ) {

        const value =
            getHeaderCI(
                headers,
                headerName
            );

        if (value === undefined) {
            continue;
        }

        const v = String(value).toLowerCase();

        if (v.includes('hit')) {
            cacheHitCount.add(1, { page: id });
            classified = true;
            break;
        }

        if (v.includes('miss')) {
            cacheMissCount.add(1, { page: id });
            classified = true;
            break;
        }
    }

    if (!classified) cacheUnknownCount.add(1, { page: id });

    // ---------------------------------------------------------------------------
    // Age
    // ---------------------------------------------------------------------------

    const ageValue = getHeaderCI(headers, cfg.ageHeader);

    if (ageValue !== undefined) {
        const ageNum = parseFloat(ageValue);
        if (!isNaN(ageNum)) cacheAge.add(ageNum, { page: id });
    }

    // ---------------------------------------------------------------------------
    // Backend response indicator
    // ---------------------------------------------------------------------------

    const backendValue = getHeaderCI(headers, cfg.backendHeader);

    if (backendValue !== undefined && String(backendValue).toLowerCase() === 'true') {
        backendResponseCount.add(1, { page: id });
    }

    // ---------------------------------------------------------------------------
    // Cache response indicator
    // ---------------------------------------------------------------------------

    const cacheRespValue = getHeaderCI(headers, cfg.cacheResponseHeader);

    if (cacheRespValue !== undefined && String(cacheRespValue).toLowerCase() === 'true') {
        cacheResponseCount.add(1, { page: id });
    }
}

// =============================================================================
// 14. REPORTING HELPERS
// =============================================================================

function metric(data, name) {

    return (
        data.metrics &&
        data.metrics[name]
    ) || null;
}

function trendStat(m, stat, fallback = 0) {
    return (m && m.values && m.values[stat] !== undefined) ? m.values[stat] : fallback;
}

function counterCount(m, fallback = 0) {
    return (m && m.values && m.values.count !== undefined) ? m.values.count : fallback;
}

function rateValue(m, fallback = 0) {
    return (m && m.values && m.values.rate !== undefined) ? m.values.rate : fallback;
}

function gaugeMax(m, fallback = 0) {
    return (m && m.values && m.values.max !== undefined) ? m.values.max : fallback;
}

function pct(part, whole) {
    return whole > 0 ? (part / whole) * 100 : 0;
}

function fmt(n, decimals = 2) {
    if (n === null || n === undefined || isNaN(n)) return 'n/a';
    return Number(n).toFixed(decimals);
}

function esc(str) {
    return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

function fmtBytes(n) {
    if (n === null || n === undefined || isNaN(n)) return 'n/a';
    const num = Number(n);
    if (num < 1024) return `${num.toFixed(0)} B`;
    if (num < 1024 * 1024) return `${(num / 1024).toFixed(2)} KB`;
    return `${(num / (1024 * 1024)).toFixed(2)} MB`;
}

// =============================================================================
// 15. BUILD PAGE METRICS FOR A SPECIFIC PHASE
// =============================================================================
//
// THIS is the main reporting change.
//
// Every page is now explicitly scoped to:
//
//     scenario = phase
//     url      = page
//
// Therefore:
//
// Phase 1000 TPS
//   Homepage response metrics
//   FAQ response metrics
//   ContactUs response metrics
//
// Phase 5000 TPS
//   Homepage response metrics
//   FAQ response metrics
//   ContactUs response metrics
//
// etc.
// =============================================================================

function buildPhasePageRow(
    data,
    phase,
    urlConf
) {

    const phaseName =
        phase.name;

    const id =
        urlConf.id;

    // ---------------------------------------------------------------------------
    // Phase + URL request count
    // ---------------------------------------------------------------------------

    const reqs =
        counterCount(
            metric(
                data,
                phaseUrlTag(
                    'http_reqs',
                    phaseName,
                    id
                )
            )
        );

    // ---------------------------------------------------------------------------
    // Phase + URL response time
    // ---------------------------------------------------------------------------

    const durationM =
        metric(
            data,
            phaseUrlTag(
                'http_req_duration',
                phaseName,
                id
            )
        );

    // ---------------------------------------------------------------------------
    // Phase + URL status classes
    // ---------------------------------------------------------------------------

    const classCounts = {};

    STATUS_CLASSES.forEach((cls) => {

        classCounts[cls] =
            counterCount(
                metric(
                    data,
                    phaseUrlClassTag(
                        'status_class_count',
                        phaseName,
                        id,
                        cls
                    )
                )
            );
    });

    const failed =
        classCounts['4xx'] +
        classCounts['5xx'] +
        classCounts['err'];

    const passed =
        Math.max(
            0,
            reqs - failed
        );

    const errorPct =
        pct(
            failed,
            reqs
        );

    return {

        id,

        name:
            urlConf.name || id,

        url:
            urlConf.url,

        count:
            reqs,

        avg:
            trendStat(
                durationM,
                'avg'
            ),

        min:
            trendStat(
                durationM,
                'min'
            ),

        p90:
            trendStat(
                durationM,
                'p(90)'
            ),

        p95:
            trendStat(
                durationM,
                'p(95)'
            ),

        p99:
            trendStat(
                durationM,
                'p(99)'
            ),

        max:
            trendStat(
                durationM,
                'max'
            ),

        passed,

        failed,

        errorPct,

        classCounts,

        status:
            errorPct <=
                CONFIG.reporting.maxErrorRatePct
                ? 'PASS'
                : 'FAIL',
    };
}

// =============================================================================
// 16. BUILD PHASE ROW
// =============================================================================
//
// Each phase contains its own pages[] collection.
//
// This gives the report hierarchy:
//
// phase
//   -> TPS
//   -> phase status
//   -> pages
//        -> Homepage
//        -> FAQ
//        -> ContactUs
// =============================================================================

function buildPhaseRow(
    data,
    phase
) {

    const name =
        phase.name;

    // ---------------------------------------------------------------------------
    // Phase request count
    // ---------------------------------------------------------------------------

    const reqs =
        counterCount(
            metric(
                data,
                phaseTag(
                    'http_reqs',
                    name
                )
            )
        );

    // ---------------------------------------------------------------------------
    // Phase response time
    //
    // Kept for overall phase-level context.
    // Page-level values are shown below it.
    // ---------------------------------------------------------------------------

    const durationM =
        metric(
            data,
            phaseTag(
                'http_req_duration',
                name
            )
        );

    // ---------------------------------------------------------------------------
    // Phase status classes
    // ---------------------------------------------------------------------------

    const classCounts = {};

    STATUS_CLASSES.forEach((cls) => {

        classCounts[cls] =
            counterCount(
                metric(
                    data,
                    phaseClassTag(
                        'status_class_count',
                        name,
                        cls
                    )
                )
            );
    });

    const failed =
        classCounts['4xx'] +
        classCounts['5xx'] +
        classCounts['err'];

    const passed =
        Math.max(
            0,
            reqs - failed
        );

    const errorPct =
        pct(
            failed,
            reqs
        );

    // ---------------------------------------------------------------------------
    // Phase duration / actual TPS
    // ---------------------------------------------------------------------------

    const durationSeconds =
        parseDurationToSeconds(
            phase.duration
        );

    const actualTps =
        durationSeconds > 0
            ? reqs / durationSeconds
            : 0;

    // ---------------------------------------------------------------------------
    // Dropped iterations
    // ---------------------------------------------------------------------------

    const dropped =
        counterCount(
            metric(
                data,
                phaseTag(
                    'dropped_iterations',
                    name
                )
            )
        );

    // ---------------------------------------------------------------------------
    // TPS achievement
    // ---------------------------------------------------------------------------

    const tpsAchievementPct =
        pct(
            actualTps,
            phase.targetTPS
        );

    const tpsOk =
        tpsAchievementPct >=
        CONFIG.reporting.minTpsAchievementPct;

    const errOk =
        errorPct <=
        CONFIG.reporting.maxErrorRatePct;

    const dropOk =
        dropped === 0;

    // ---------------------------------------------------------------------------
    // PAGE METRICS INSIDE THIS PHASE
    // ---------------------------------------------------------------------------

    const pages =
        CONFIG.urls.map(
            (urlConf) =>
                buildPhasePageRow(
                    data,
                    phase,
                    urlConf
                )
        );

    const networkMetrics = buildPhaseNetworkMetrics(data, name);

    return {
        phase: name,
        targetTps: phase.targetTPS,
        actualTps,
        tpsAchievementPct,
        duration: phase.duration,
        avg: trendStat(durationM, 'avg'),
        min: trendStat(durationM, 'min'),
        p90: trendStat(durationM, 'p(90)'),
        p95: trendStat(durationM, 'p(95)'),
        p99: trendStat(durationM, 'p(99)'),
        max: trendStat(durationM, 'max'),
        count: reqs,
        passed,
        failed,
        errorPct,
        classCounts,
        dropped,
        tpsOk,
        errOk,
        dropOk,
        status: tpsOk && errOk && dropOk ? 'PASS' : 'FAIL',
        pages,
        networkMetrics,
    };
}

// =============================================================================
// 16b. NATIVE NETWORK METRICS FOR A PHASE
// =============================================================================
//
// Pulls back out the native k6 metrics that were already registered as
// phase-scoped submetrics in buildThresholds(). No new metric objects are
// created here -- this only reads data k6 was already collecting.
// =============================================================================

function buildPhaseNetworkMetrics(
    data,
    phaseName
) {

    const trends =
        NETWORK_TREND_METRICS.map((m) => {

            const met =
                metric(
                    data,
                    phaseTag(
                        m.key,
                        phaseName
                    )
                );

            return {

                key:
                    m.key,

                label:
                    m.label,

                avg:
                    trendStat(met, 'avg'),

                min:
                    trendStat(met, 'min'),

                p90:
                    trendStat(met, 'p(90)'),

                p95:
                    trendStat(met, 'p(95)'),

                p99:
                    trendStat(met, 'p(99)'),

                max:
                    trendStat(met, 'max'),
            };
        });

    const dataSentBytes =
        counterCount(
            metric(
                data,
                phaseTag('data_sent', phaseName)
            )
        );

    const dataReceivedBytes =
        counterCount(
            metric(
                data,
                phaseTag('data_received', phaseName)
            )
        );

    const iterDurM =
        metric(
            data,
            phaseTag('iteration_duration', phaseName)
        );

    const iterationDuration = {

        avg:
            trendStat(iterDurM, 'avg'),

        min:
            trendStat(iterDurM, 'min'),

        p90:
            trendStat(iterDurM, 'p(90)'),

        p95:
            trendStat(iterDurM, 'p(95)'),

        p99:
            trendStat(iterDurM, 'p(99)'),

        max:
            trendStat(iterDurM, 'max'),
    };

    const httpReqFailedM =
        metric(
            data,
            phaseTag('http_req_failed', phaseName)
        );

    const httpReqFailedPct =
        rateValue(httpReqFailedM) * 100;

    return {

        trends,

        dataSentBytes,

        dataReceivedBytes,

        iterationDuration,

        httpReqFailedPct,
    };
}

// =============================================================================
// 17. CACHE REPORT
// =============================================================================

function buildCacheRow(data, urlConf) {
    const id = urlConf.id;
    const strategy = CACHE_STRATEGY[urlConf.provider] || 'header';

    if (strategy === 'not-applicable') {
        return { id, name: urlConf.name || id, url: urlConf.url, strategy, note: 'No CDN in front of this URL -- not applicable.' };
    }

    if (strategy === 'not-observable') {
        return { id, name: urlConf.name || id, url: urlConf.url, strategy, note: 'GCP Cloud CDN does not expose cache status to the client. Check cache_result in Cloud Monitoring / Cloud Logging for the test time window.' };
    }

    const hit = counterCount(metric(data, urlTag('cache_hit_count', id)));
    const miss = counterCount(metric(data, urlTag('cache_miss_count', id)));
    const unknown = counterCount(metric(data, urlTag('cache_unknown_count', id)));
    const backend = counterCount(metric(data, urlTag('backend_response_count', id)));
    const cacheResp = counterCount(metric(data, urlTag('cache_response_count', id)));
    const avgAge = trendStat(metric(data, urlTag('cache_age', id)), 'avg');
    const classifiable = hit + miss;

    return { id, name: urlConf.name || id, url: urlConf.url, strategy, hit, miss, unknown, hitPct: pct(hit, classifiable), backend, cacheResp, avgAge };
}

// 19. HANDLE SUMMARY

//
// Runs once after the test.

export function handleSummary(data) {

    const durationMs =
        data.state &&
            data.state.testRunDurationMs
            ? data.state.testRunDurationMs
            : 0;

    const durationSec =
        durationMs / 1000;

    const endTime =
        new Date();

    const startTime =
        new Date(
            endTime.getTime() -
            durationMs
        );

    // ===========================================================================
    // OVERALL REQUESTS
    // ===========================================================================

    const overallReqs =
        counterCount(
            metric(
                data,
                'http_reqs'
            )
        );

    const overallDurationM =
        metric(
            data,
            'http_req_duration'
        );

    // ===========================================================================
    // OVERALL STATUS CLASSES
    // ===========================================================================
    //
    // Sum the bounded URL-level class metrics.
    // This does NOT scan individual requests.
    // ===========================================================================

    const overallClassCounts = {};

    STATUS_CLASSES.forEach((cls) => {

        overallClassCounts[cls] =
            CONFIG.urls.reduce(
                (sum, u) => {

                    return sum +
                        counterCount(
                            metric(
                                data,
                                urlClassTag(
                                    'status_class_count',
                                    u.id,
                                    cls
                                )
                            )
                        );
                },
                0
            );
    });

    const overallFailed =
        overallClassCounts['4xx'] +
        overallClassCounts['5xx'] +
        overallClassCounts['err'];

    const overallPassed =
        Math.max(
            0,
            overallReqs -
            overallFailed
        );

    const overallErrorPct =
        pct(
            overallFailed,
            overallReqs
        );

    const overallDropped =
        counterCount(
            metric(
                data,
                'dropped_iterations'
            )
        );

    // ===========================================================================
    // TARGET TPS
    // ===========================================================================

    const overallTargetTps =
        CONFIG.phases.reduce(
            (max, p) =>
                Math.max(
                    max,
                    p.targetTPS
                ),
            0
        );

    // ===========================================================================
    // OVERALL ACTUAL TPS
    // ===========================================================================
    //
    // This is informational only because multiple phases can have different
    // target rates.
    //
    // The authoritative TPS assessment is in the phase table.
    // ===========================================================================

    const overallActualTps =
        durationSec > 0
            ? overallReqs / durationSec
            : 0;

    // ===========================================================================
    // OVERALL ACCEPTANCE
    // ===========================================================================

    const errOk =
        overallErrorPct <=
        CONFIG.reporting.maxErrorRatePct;

    const dropOk =
        overallDropped === 0;

    // For multi-phase tests, phase-level TPS achievement is authoritative.
    const allPhasesTpsOk =
        CONFIG.phases.every((phase) => {

            const phaseReqs =
                counterCount(
                    metric(
                        data,
                        phaseTag(
                            'http_reqs',
                            phase.name
                        )
                    )
                );

            const phaseDuration =
                parseDurationToSeconds(
                    phase.duration
                );

            const phaseActualTps =
                phaseDuration > 0
                    ? phaseReqs /
                    phaseDuration
                    : 0;

            const achievement =
                pct(
                    phaseActualTps,
                    phase.targetTPS
                );

            return achievement >=
                CONFIG.reporting.minTpsAchievementPct;
        });

    const overallStatus =
        allPhasesTpsOk &&
            errOk &&
            dropOk
            ? 'PASS'
            : 'FAIL';

    // ===========================================================================
    // FINAL SUMMARY OBJECT
    // ===========================================================================

    const summary = {

        meta: {

            testType:
                CONFIG.testType,

            startTime:
                startTime.toISOString(),

            endTime:
                endTime.toISOString(),

            durationSec,

            pageCount:
                CONFIG.urls.length,

            phaseCount:
                CONFIG.phases.length,
        },

        // -------------------------------------------------------------------------
        // OVERALL
        // -------------------------------------------------------------------------

        overall: {

            count:
                overallReqs,

            passed:
                overallPassed,

            failed:
                overallFailed,

            errorPct:
                overallErrorPct,

            targetTps:
                overallTargetTps,

            actualTps:
                overallActualTps,

            dropped:
                overallDropped,

            avg:
                trendStat(
                    overallDurationM,
                    'avg'
                ),

            min:
                trendStat(
                    overallDurationM,
                    'min'
                ),

            p90:
                trendStat(
                    overallDurationM,
                    'p(90)'
                ),

            p95:
                trendStat(
                    overallDurationM,
                    'p(95)'
                ),

            p99:
                trendStat(
                    overallDurationM,
                    'p(99)'
                ),

            max:
                trendStat(
                    overallDurationM,
                    'max'
                ),

            // vus_max is a Gauge, so use max rather than Counter.count.
            vusMax:
                gaugeMax(
                    metric(
                        data,
                        'vus_max'
                    )
                ),

            classCounts:
                overallClassCounts,

            status:
                overallStatus,

            tpsOk:
                allPhasesTpsOk,

            errOk,

            dropOk,
        },

        // -------------------------------------------------------------------------
        // PHASES
        //
        // Each phase contains pages[].
        // -------------------------------------------------------------------------

        phases:
            CONFIG.phases.map(
                (phase) =>
                    buildPhaseRow(
                        data,
                        phase
                    )
            ),

        // -------------------------------------------------------------------------
        // CACHE
        // -------------------------------------------------------------------------

        cache:
            CONFIG.urls.map(
                (url) =>
                    buildCacheRow(
                        data,
                        url
                    )
            ),

        // -------------------------------------------------------------------------
        // Native aggregate metrics.
        //
        // Retained for future dashboarding.
        // -------------------------------------------------------------------------

        nativeMetrics:
            data.metrics,
    };

    return {

        'k6-report.html':
            buildHtmlReport(
                summary
            ),

        'k6-summary.json':
            JSON.stringify(
                summary,
                null,
                2
            ),

        stdout:
            textSummary(
                data,
                {
                    indent: ' ',
                    enableColors: true,
                }
            ) +
            `\nTest complete. Overall status: ${summary.overall.status}. See k6-report.html and k6-summary.json.\n`,
    };
}

function num(value, fallback = 0) {
    const n = Number(value);
    return Number.isFinite(n) ? n : fallback;
}

function int(value, fallback = 0) {
    return Math.round(num(value, fallback));
}

function pct(value, digits = 2) {
    return `${num(value).toFixed(digits)}%`;
}

function ms(value, digits = 2) {
    return `${num(value).toFixed(digits)} ms`;
}

function formatNumber(value, digits = 2) {
    const n = num(value);

    return new Intl.NumberFormat('en-US', {
        minimumFractionDigits: digits,
        maximumFractionDigits: digits
    }).format(n);
}

function formatInteger(value) {
    return new Intl.NumberFormat('en-US', {
        maximumFractionDigits: 0
    }).format(int(value));
}

function escapeHtml(value) {
    return String(value ?? '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

function safeJson(value) {
    return JSON.stringify(value)
        .replace(/</g, '\\u003c')
        .replace(/>/g, '\\u003e')
        .replace(/&/g, '\\u0026');
}

function safeId(value) {
    return String(value ?? '')
        .replace(/[^a-zA-Z0-9_-]/g, '_')
        .replace(/^(\d)/, '_$1');
}

function average(values) {
    const valid = values.filter(Number.isFinite);

    if (!valid.length) {
        return 0;
    }

    return valid.reduce((a, b) => a + b, 0) / valid.length;
}

function maxBy(items, selector) {
    if (!items.length) {
        return null;
    }

    return items.reduce((best, item) => {
        return selector(item) > selector(best) ? item : best;
    });
}

function minBy(items, selector) {
    if (!items.length) {
        return null;
    }

    return items.reduce((best, item) => {
        return selector(item) < selector(best) ? item : best;
    });
}

// =============================================================================
// STATUS CLASS TABLE
// =============================================================================

function buildStatusClassTable() {
    const total = Object.values(overallStatusClasses).reduce(
        (a, b) => a + b,
        0
    );

    const rows = Object.entries(overallStatusClasses)
        .map(([status, count]) => {
            const percentage =
                total > 0 ? (count / total) * 100 : 0;

            return `
                <tr>
                    <td>${escapeHtml(status)}</td>
                    <td>${formatInteger(count)}</td>
                    <td>${pct(percentage)}</td>
                </tr>
            `;
        })
        .join('');

    return `
        <div class="table-wrapper compact">
            <table>
                <thead>
                    <tr>
                        <th>Status Class</th>
                        <th>Count</th>
                        <th>Percentage</th>
                    </tr>
                </thead>
                <tbody>${rows}</tbody>
            </table>
        </div>
    `;
}

// =============================================================================
// ACTUAL STATUS CODE TABLE
// =============================================================================

function buildActualStatusTable() {
    const entries = Object.entries(actualStatusCodes)
        .sort((a, b) => Number(a[0]) - Number(b[0]));

    if (!entries.length) {
        return `
            <div class="empty-state">
                Individual HTTP status-code counts were not present
                in the supplied summary JSON.
                <br>
                The report therefore uses the available 2xx/3xx/4xx/5xx
                class counts instead of inventing individual codes.
            </div>
        `;
    }

    const total = entries.reduce(
        (sum, [, count]) => sum + count,
        0
    );

    const rows = entries
        .map(([code, count]) => `
            <tr>
                <td><strong>${escapeHtml(code)}</strong></td>
                <td>${formatInteger(count)}</td>
                <td>${pct(total ? (count / total) * 100 : 0)}</td>
            </tr>
        `)
        .join('');

    return `
        <div class="table-wrapper compact">
            <table>
                <thead>
                    <tr>
                        <th>HTTP Code</th>
                        <th>Count</th>
                        <th>Percentage</th>
                    </tr>
                </thead>
                <tbody>${rows}</tbody>
            </table>
        </div>
    `;
}

const transactionErrors = pages.map(page => {
    let count = 0;
    let total = 0;

    for (const phase of phases) {
        const p = (phase.pages || []).find(
            x => (x.id || x.name) === page.id
        );

        if (p) {
            total += int(p.count);
            count += int(p.failed);
        }
    }

    return {
        name: page.name,
        errors: count,
        total
    };
});

// RESOURCE SECTIONS
// =============================================================================

function buildResourceSections(pageId) {
    const resourceRows = [];

    for (const phase of sortedPhases) {
        const page = (phase.pages || []).find(
            p => (p.id || p.name) === pageId
        );

        if (!page) {
            continue;
        }

        for (const resource of getPageResources(page)) {
            resourceRows.push({
                phase,
                resource
            });
        }
    }

    if (!resourceRows.length) {
        return '';
    }

    const rows = resourceRows
        .map(({ phase, resource }) => {
            const name =
                resource.name ||
                resource.url ||
                resource.path ||
                resource.id ||
                'Resource';

            return `
                <tr>
                    <td>${formatNumber(phase.targetTps, 0)} TPS</td>
                    <td class="resource-url">

                    <td>${ms(resource.p95)}</td>
                    <td>${ms(resource.p99)}</td>
                    <td>${ms(resource.max)}</td>
                    <td>${pct(resource.errorPct)}</td>
                </tr>
            `;
        })
        .join('');

    return `
        <details class="accordion resource-accordion">
            <summary>
                Embedded Resources
            </summary>

            <div class="accordion-body">
                <div class="table-wrapper">
                    <table>

}
